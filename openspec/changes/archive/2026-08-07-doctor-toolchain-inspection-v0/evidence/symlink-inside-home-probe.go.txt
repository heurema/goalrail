// Probe: does a confined root follow a symbolic link whose target stays inside
// it? Retained as evidence for the round-6 finding.
package main

import (
	"fmt"
	"io"
	"os"
	"path/filepath"
)

func main() {
	home := filepath.Join(os.TempDir(), "goalrail-symlink-probe")
	os.RemoveAll(home)
	planted := filepath.Join(home, "planted", "darwin_arm64")
	os.MkdirAll(planted, 0o755)
	os.WriteFile(filepath.Join(planted, "manifest.json"), []byte("PLANTED-TREE"), 0o644)

	bundles := filepath.Join(home, ".local", "share", "goalrail", "bundles")
	os.MkdirAll(bundles, 0o755)
	// A relative target, which stays inside the home root and is therefore not
	// an escape by the root's own rule.
	if err := os.Symlink("../../../../planted", filepath.Join(bundles, "v0.2.0")); err != nil {
		fmt.Println("symlink:", err)
		return
	}

	root, _ := os.OpenRoot(home)
	sub, err := root.OpenRoot(".local/share/goalrail/bundles/v0.2.0/darwin_arm64")
	fmt.Printf("relative symlink inside the root followed: %t (err: %v)\n", sub != nil, err)
	if sub != nil {
		if f, openErr := sub.Open("manifest.json"); openErr == nil {
			body, _ := io.ReadAll(f)
			fmt.Printf("bytes read from the planted tree: %q\n", body)
		}
	}
}
